This guarantees that any cluster recurring across the two distinct runs (notably `unsupported_future__refuse_expected_but_answered`) reaches `seen_count == 2` per 11-SPEC.md §3 and becomes eligible for promotion in Task 6.

```python
import json
from pathlib import Path
from evals.discovery.failure_backlog import load_backlog, append_or_update, save_backlog
from evals.discovery.triage_rubric import classify
from evals.discovery.iteration_runner import compare_runs, write_diff

# --- Step A: seed the backlog from the Phase 10 live run (FIRST distinct run_id) ---
phase10_run_id = "2026-04-29_09-51-54__synthetic_phase10_first"
phase10_dir = Path(f"evals/runs/{phase10_run_id}")
phase10_clusters = json.loads((phase10_dir / "failure_clusters.json").read_text(encoding="utf-8"))

backlog = load_backlog()
for cluster in phase10_clusters["clusters"]:
    backlog = append_or_update(backlog, cluster, run_id=phase10_run_id)
# Do NOT classify or set status yet — Step B may revise via the recurring-cluster path.

# --- Step B: update from the new unsupported_future_v1 run (SECOND distinct run_id) ---
new_run_dir = Path("evals/runs/<new run dir>")  # captured from Task 4 output
new_run_id = new_run_dir.name
assert new_run_id != phase10_run_id, "new run_id must be distinct from the Phase 10 seed run_id"
new_clusters = json.loads((new_run_dir / "failure_clusters.json").read_text(encoding="utf-8"))

for cluster in new_clusters["clusters"]:
    backlog = append_or_update(backlog, cluster, run_id=new_run_id)
    entry = next(e for e in backlog["entries"] if e["failure_signature"] == cluster["failure_signature"])
    entry["recommended_action"] = classify(cluster)
    entry["status"] = "triaged"

# --- Step C: classify any seeded Phase 10 entries that did not recur in the new run ---
for entry in backlog["entries"]:
    if entry.get("recommended_action") is None:
        # synthesize a minimal cluster dict for the deterministic classifier
        synthetic_cluster = {
            "category": entry["category"],
            "guard_evidence": entry["guard_evidence"],
        }
        entry["recommended_action"] = classify(synthetic_cluster)
        entry["status"] = "triaged"

save_backlog(backlog)

# --- Step D: write the diff JSON (newer vs older Phase 10 run) ---
diff = compare_runs(phase10_dir, new_run_dir)
write_diff(diff, new_run_dir)
```
